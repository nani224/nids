Q1 — Alerts by Severity GROUP BY + COUNT

FROM ALERT a JOIN INTRUSION_RULE r ON a.rule_id = r.rule_id

Q2 — Top Triggered Rules ORDER BY + LIMIT
SELECT r.rule_name, COUNT(a.alert_id) AS cnt
FROM ALERT a JOIN INTRUSION_RULE r ON a.rule_id = r.rule_id
GROUP BY r.rule_id ORDER BY cnt DESC;

Q3 — Device Threat Report Multi-JOIN + GROUP BY
SELECT d.device_name, d.ip_address, COUNT(a.alert_id) AS alert_count
FROM NETWORK_DEVICE d JOIN NETWORK_LOG l ON d.device_id = l.device_id
JOIN ALERT a ON l.log_id = a.log_id
GROUP BY d.device_id ORDER BY alert_count DESC;
